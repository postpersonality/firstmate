#!/usr/bin/env bash
# Bounded offline checks; every fixture path stays under $OWN/fx.
set -u
ROOT=$1 OWN=$2
FX="$OWN/fx"; mkdir "$FX" || exit 1
TASKS_AXI_BIN=$(command -v tasks-axi)
checks=0 failed=0
check() { # <label> <expected: ok|refuse> <cmd...>
  local label=$1 want=$2 rc; shift 2
  checks=$((checks+1))
  printf '$ %s\n' "$*"
  "$@" > "$FX/last.out" 2>&1; rc=$?
  sed 's/^/  | /' "$FX/last.out" | head -6
  if { [ "$want" = ok ] && [ "$rc" = 0 ]; } || { [ "$want" = refuse ] && [ "$rc" != 0 ]; }; then
    printf 'PASS [exit %s]: %s\n\n' "$rc" "$label"
  else
    failed=$((failed+1)); printf 'FAIL [exit %s]: %s\n\n' "$rc" "$label"
  fi
}
make_home() { # <name> <kind>
  local home="$FX/$1" id=sample-outcome
  mkdir -p "$home/data/$id" "$home/state" "$home/config" "$home/projects" "$home/fakebin"
  cp "$ROOT/.tasks.toml" "$home/.tasks.toml"
  printf '## In flight\n\n## Queued\n\n## Done\n' > "$home/data/backlog.md"
  for t in tmux treehouse no-mistakes gh gh-axi; do printf '#!/usr/bin/env bash\nexit 0\n' > "$home/fakebin/$t"; chmod +x "$home/fakebin/$t"; done
  (cd "$home" && tasks-axi add "$id" "Working evaluator on the saved corpus" >/dev/null && tasks-axi start "$id" >/dev/null)
  printf '%s\n' "window=firstmate:fm-$id" "worktree=$home/projects/missing-$id" "project=$home/projects/sample" \
    harness=codex "kind=$2" "mode=$2" "spawn_gen=fixture-$id" completion_required=1 > "$home/state/$id.meta"
  printf '# Corpus and calibration plan\n' > "$home/data/$id/report.md"
  printf 'done: bounded milestone saved\n' > "$home/state/$id.status"
  printf '%s\n' "$home"
}
cap() { local home=$1; shift; PATH="$home/fakebin:$PATH" REAL_TASKS_AXI="$TASKS_AXI_BIN" FM_HOME="$home" FM_STATE_OVERRIDE="$home/state" FM_DATA_OVERRIDE="$home/data" FM_CONFIG_OVERRIDE="$home/config" "$ROOT/bin/fm-captain-hold.sh" "$@"; }
tdn() { local home=$1; PATH="$home/fakebin:$PATH" FM_ROOT_OVERRIDE="$ROOT" FM_HOME="$home" FM_STATE_OVERRIDE="$home/state" FM_DATA_OVERRIDE="$home/data" FM_CONFIG_OVERRIDE="$home/config" "$ROOT/bin/fm-teardown.sh" "$2"; }
